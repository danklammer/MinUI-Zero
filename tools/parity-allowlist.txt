# Declared cross-platform divergences — tools/check-parity.sh fails anything not listed here.
# Format: <pak>:<key> (pak = Emus pak dir name, or `system` for system.cfg).
# Every entry needs a reason. A stale entry (values now match) also fails the check.

# Panel geometry: 640x480 makes NES-at-Aspect 2.5x horizontal -> scroll shimmer (Contra report);
# 1024x768 gives a clean 4.0x, and Native there would shrink the image. See FC.pak comments + D-log.
FC.pak:minarch_screen_scaling
GB.pak:minarch_screen_scaling
GBC.pak:minarch_screen_scaling
system:minarch_screen_scaling

# MMP is dual-core (threading Off per-pak on top of the system lock) and has no analog sticks
# (no DualShock default).
PS.pak:minarch_threading
PS.pak:minarch_gamepad_type

# MMP LOCKS sharpness: PLAT_setSharpness ignores the value there (platform.c — one scaler family,
# effects carry the look), so the menu entry would be a no-op choice. tg5040 keeps it unlocked.
system:minarch_screen_sharpness

# Present policy follows the PANEL, and the panels differ. The MMP's is 59.6720Hz (measured), below
# a 60fps core, so Lenient silently drops the surplus frame; Strict + MINARCH_PANEL_FPS + the page
# ownership state machine eliminate it. tg5040 presents via SDL2 PRESENTVSYNC, which has no
# coalescing path and lets the driver own sync, so it has neither the bug nor a reason to change.
system:minarch_prevent_tearing

# supafaust affinity masks name CPUs 2-3, which do not exist on the dual-A7 MMP.
SUPA.pak:supafaust_thread_affinity_emu
SUPA.pak:supafaust_thread_affinity_ppu
